# Interpolation function

def interpolate_signals(df, signal_cols, group_col="subject_id"):
    df = df.copy()

    for col in signal_cols:
        df[col] = df.groupby(group_col)[col].transform(
            lambda s: s.interpolate(method="linear", limit_direction="both")
        )

    return df

def apply_savgol_filter(df, signal_cols, window_length=11, polyorder=2, group_col="subject_id"):
    df = df.copy()

    for subject_id, subject_df in df.groupby(group_col):
        idx = subject_df.index

        for col in signal_cols:
            series = subject_df[col].values

            wl = min(window_length, len(series))
            if wl % 2 == 0:
                wl -= 1

            if wl > polyorder:
                df.loc[idx, col] = savgol_filter(series, window_length=wl, polyorder=polyorder)

    return df

# Window segmentation helper

def segment_windows(df, signal_cols, window_size, overlap_ratio=0.5, group_col="subject_id", label_col="label"):
    X_windows = []
    y_windows = []
    subjects = []

    step = int(window_size * (1 - overlap_ratio))
    step = max(step, 1)

    for subject_id, subject_df in df.groupby(group_col):
        subject_df = subject_df.reset_index(drop=True)

        for start in range(0, len(subject_df) - window_size + 1, step):
            end = start + window_size
            window_df = subject_df.iloc[start:end]

            # majority-vote label
            window_label = window_df[label_col].mode().iloc[0]

            # optionally skip highly mixed windows
            X_windows.append(window_df[signal_cols].values)
            y_windows.append(window_label)
            subjects.append(subject_id)

    return np.array(X_windows), np.array(y_windows), np.array(subjects)

# Time-domain features for one 1D signal

def extract_time_domain_features(signal_1d):
    signal_1d = np.asarray(signal_1d)

    mean_val = np.mean(signal_1d)
    std_val = np.std(signal_1d)
    rms_val = np.sqrt(np.mean(signal_1d ** 2))
    zcr_val = np.sum(np.diff(np.sign(signal_1d)) != 0) / max(len(signal_1d) - 1, 1)

    return {
        "mean": mean_val,
        "std": std_val,
        "rms": rms_val,
        "zcr": zcr_val
    }

# Frequency-domain features for one 1D signal

def extract_frequency_domain_features(signal_1d, sampling_rate=50):
    signal_1d = np.asarray(signal_1d)
    n = len(signal_1d)

    yf = np.abs(fft(signal_1d))
    xf = fftfreq(n, d=1 / sampling_rate)

    pos_mask = xf > 0
    xf = xf[pos_mask]
    yf = yf[pos_mask]

    if len(yf) == 0:
        return {
            "dominant_freq": 0.0,
            "spectral_energy": 0.0,
            "spectral_entropy": 0.0
        }

    dominant_freq = xf[np.argmax(yf)]
    spectral_energy = np.sum(yf ** 2)

    prob = yf / np.sum(yf) if np.sum(yf) != 0 else np.ones_like(yf) / len(yf)
    spectral_entropy = entropy(prob)

    return {
        "dominant_freq": dominant_freq,
        "spectral_energy": spectral_energy,
        "spectral_entropy": spectral_entropy
    }

# Convert one multichannel window into one feature dictionary

def window_to_feature_vector(window_array, signal_cols, sampling_rate=50, include_optional=False):
    feature_dict = {}

    for i, col in enumerate(signal_cols):
        sig = window_array[:, i]

        td = extract_time_domain_features(sig)
        fd = extract_frequency_domain_features(sig, sampling_rate=sampling_rate)

        for k, v in td.items():
            feature_dict[f"{col}_{k}"] = v

        for k, v in fd.items():
            feature_dict[f"{col}_{k}"] = v

        if include_optional:
            extra = extract_optional_features(sig)
            for k, v in extra.items():
                feature_dict[f"{col}_{k}"] = v

    return feature_dict

Model Results & Descriptions¶

Model Descriptions¶

Decision Tree is a simple rule-based classifier that splits the feature space by the most informative features at each node. It is fast and interpretable but prone to overfitting on complex datasets.

Support Vector Machine (SVM) finds the optimal hyperplane that separates classes in high-dimensional space. Using an RBF kernel allows it to capture nonlinear boundaries between activity classes.

Naive Bayes assumes all features are conditionally independent given the class label. While this assumption rarely holds in practice, it serves as a fast and useful baseline.

Random Forest is an ensemble of decision trees trained on random feature subsets. It reduces overfitting through averaging and generally produces robust results on structured feature data.

AdaBoost is a boosting ensemble that iteratively focuses on misclassified samples, assigning higher weight to harder examples with each round. It improves weak learners sequentially.

XGBoost is a gradient boosting framework optimized for speed and performance. It builds trees sequentially to correct previous errors and consistently performs well on tabular, feature-engineered data.

Modeling notes¶

  • Best model (macro F1): XGBoost at 0.82 macro F1, 96.4% accuracy.
  • Honest headline: Weighted F1 of 0.962 is inflated by class imbalance — walking alone accounts for 48% of the test set. Macro F1 is the correct metric for comparing across all 7 classes.
  • Naive Bayes: Posted the lowest macro F1 (0.53). Its high weighted precision (0.94) is misleading — this is dominated by near-perfect precision on the four majority classes. Per-class inspection shows precision of 0.01 on stairs descending and 0.16 on shuffling and stairs ascending. The Gaussian independence assumption likely fails here: accelerometer features derived from the same sensor axes are physically correlated, violating the model's core assumption.
  • AdaBoost: Low macro F1 (0.58) particularly on stairs descending (F1=0.04) and shuffling (F1=0.02). This is a known failure mode of SAMME.R boosting under severe multiclass imbalance — the algorithm's reweighting collapses on classes with very few support samples.
  • Tree-based ensembles (RF, XGBoost): Outperform SVM and single-tree methods because they handle nonlinear feature interactions without distribution assumptions and are robust to the feature scale differences across time and frequency domain features.
  • Limitation: Standard 80/20 random split with 50% window overlap introduces near-duplicate windows across train and test (adjacent windows share 50 samples). This inflates all reported metrics relative to a subject-independent evaluation. Subject-independent performance is evaluated separately via LOSO (see Section 9).

# LOSO evaluation helper

def run_loso_evaluation(X_df, y_series, subject_series, model):
    unique_subjects = sorted(subject_series.unique())
    fold_results = []

    for test_subject in unique_subjects:
        train_idx = subject_series != test_subject
        test_idx = subject_series == test_subject

        X_train = X_df.loc[train_idx]
        X_test = X_df.loc[test_idx]
        y_train = y_series.loc[train_idx]
        y_test = y_series.loc[test_idx]

        model.fit(X_train, y_train)
        y_pred = model.predict(X_test)

        metrics = compute_metrics(y_test, y_pred)
        metrics["test_subject"] = test_subject
        fold_results.append(metrics)

    return pd.DataFrame(fold_results)

LOSO Interpretation¶

  • LOSO is harder because the model must generalize to an unseen participant with different motion style and sensor placement nuances.
  • Standard random splits may include windows from the same subject in both train and test, which inflates apparent performance.
  • The gap between standard split and LOSO quantifies subject-transfer difficulty and should guide final claims.
  • A strong final pipeline is one with stable LOSO macro-F1 across folds, not only a high single split score.
